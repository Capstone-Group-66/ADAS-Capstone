################################################################################
# File: config/dashcamnet_pgie_config.txt
# nvinfer Primary GIE configuration for NVIDIA DashCamNet (ResNet-18 Pruned)
#
# Model: TrafficCamNet (NVIDIA NGC) — used as DashCamNet equivalent
#   Classes: 0=Car, 1=Bicycle, 2=Person, 3=RoadSign
#   Architecture: ResNet-18 (pruned), input 960x544
#   Precision: FP16 (network-mode=2)
#
# Key values are locked per Phase-2 spec:
#   - network-mode  = 2  (FP16)
#   - interval      = 0  (every frame)
#   - cluster-mode  = 1  (Hardware NMS)
#
# DeepStream documentation reference:
#   https://docs.nvidia.com/metropolis/deepstream/dev-guide/text/DS_plugin_gst-nvinfer.html
################################################################################

[property]
gpu-id=0

# ─── Model engine / ONNX ────────────────────────────────────────────────────
# Point this at the DashCamNet ONNX downloaded from NGC, or the pre-built
# TensorRT engine generated by tao-converter / trt-exec.
#
# NGC download (requires free NVIDIA developer account):
#   ngc registry model download-version "nvidia/tao/dashcamnet:pruned_onnx_v1.0"
#
# After downloading, place resnet18_dashcamnet_pruned.onnx in models/ and
# let DeepStream auto-generate the .engine on first launch, OR pre-convert:
#   trtexec --onnx=models/resnet18_dashcamnet_pruned.onnx \
#           --fp16 --saveEngine=models/dashcamnet.engine
model-engine-file=models/dashcamnet.engine
onnx-file=models/resnet18_dashcamnet_pruned.onnx

# ─── Input / pre-processing ─────────────────────────────────────────────────
# DashCamNet expects 960x544 normalised RGB.
# nvstreammux will scale from 1280x720 before feeding nvinfer.
infer-dims=3;544;960
batch-size=1
maintain-aspect-ratio=1

# ─── Precision ───────────────────────────────────────────────────────────────
# 0=FP32  1=INT8  2=FP16  (spec: FP16)
network-mode=2

# ─── Inference frequency ────────────────────────────────────────────────────
# interval=0  → run inference on every single frame (spec requirement)
interval=0

# ─── Clustering (NMS) ───────────────────────────────────────────────────────
# cluster-mode=1  → Hardware NMS via OpenCV groupRectangles (fast, GPU-side)
# cluster-mode=3  → NMS post-processing (fallback if 1 not supported)
cluster-mode=1

# ─── Detection thresholds ────────────────────────────────────────────────────
threshold=0.2

# ─── Class labels ────────────────────────────────────────────────────────────
# Inline label map for DashCamNet 4-class output.
# Format: one label per line, index = line number (0-based).
labelfile-path=config/dashcamnet_labels.txt

# ─── Output tensor meta (for NvDsObjectMeta extraction in pad probe) ─────────
output-tensor-meta=0

# ─── Miscellaneous ───────────────────────────────────────────────────────────
# Parse function for DashCamNet — uses the standard detector parser.
parse-bbox-func-name=NvDsInferParseCustomResnet
custom-lib-path=/opt/nvidia/deepstream/deepstream/lib/libnvds_infer_custom_parser.so

[class-attrs-all]
nms-iou-threshold=0.45
pre-cluster-threshold=0.2
